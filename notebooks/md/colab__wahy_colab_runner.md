### 📌 Features enabled in this Colab session:
1. **Speech-to-Text (ASR):**
   - **Option A (Specialized Local GPU):** `Abu-Dju/whisper-large-v3-ar` (~2B parameters, loaded directly in Colab GPU VRAM with word-level timestamps).
   - **Option B (Fast Cloud API):** Groq Cloud API (`whisper-large-v3-turbo`) for instant cloud transcription.
2. **Quotation Detection Engine (V5):** Quran and Hadith detection with word-level candidate alignment.
3. **AI Verifier (Laya):** Loaded on the same GPU using **`ZenoAbderrahim/laya-wahy-r6-v3_mixed`** with Envelope v3 & multi-verse recitation passages.
4. **Public Web UI:** Exposes the full interactive Wahy Transcriber web interface via a secure Cloudflare Tunnel (`trycloudflare.com`).

# =============================================================================
# Step 3: Launch Wahy Transcriber & Expose Public URL via Cloudflare Tunnel
# =============================================================================
import subprocess, time, re, sys
from IPython.display import display, HTML, clear_output

# 1. Download and install cloudflared if not present
if not os.path.exists("/usr/local/bin/cloudflared") and not os.path.exists("/usr/bin/cloudflared"):
    print("⏳ Installing Cloudflare Tunnel...")
    !wget -q -nc https://github.com/cloudflare/cloudflared/releases/latest/download/cloudflared-linux-amd64.deb
    !dpkg -i cloudflared-linux-amd64.deb > /dev/null
    print("✅ Cloudflared installed.")

# 2. Kill any stale uvicorn or cloudflared processes
!pkill -f uvicorn || true
!pkill -f cloudflared || true
time.sleep(1)

# 3. Start FastAPI backend (port 8000)
backend_env = os.environ.copy()
backend_env["COLAB_GPU"] = "1"
backend_proc = subprocess.Popen(
    [sys.executable, "-m", "uvicorn", "app:app", "--host", "127.0.0.1", "--port", "8000"],
    cwd="/content/Wahy-Transcriber/backend",
    env=backend_env,
    stdout=subprocess.PIPE,
    stderr=subprocess.STDOUT,
    text=True
)
print("⏳ Starting Wahy FastAPI server...")
time.sleep(3)

# 4. Start Cloudflare Tunnel to port 8000 and capture the public URL
print("⏳ Generating secure public URL via Cloudflare Tunnel...")
cf_proc = subprocess.Popen(
    ["cloudflared", "tunnel", "--url", "http://127.0.0.1:8000"],
    stdout=subprocess.PIPE,
    stderr=subprocess.STDOUT,
    text=True
)

tunnel_url = None
for _ in range(50):
    line = cf_proc.stdout.readline()
    if not line:
        time.sleep(0.5)
        continue
    match = re.search(r"https://[a-zA-Z0-9-]+\.trycloudflare\.com", line)
    if match:
        tunnel_url = match.group(0)
        break

clear_output()
if tunnel_url:
    display(HTML(f"""
    <div style="background: linear-gradient(135deg, #0f172a, #1e293b); padding: 24px; border-radius: 16px; border: 2px solid #38bdf8; font-family: system-ui, sans-serif; color: #f8fafc; max-width: 700px; box-shadow: 0 10px 25px -5px rgba(0,0,0,0.5);">
      <div style="display: flex; align-items: center; gap: 12px; margin-bottom: 12px;">
        <span style="font-size: 32px;">🎉</span>
        <div>
          <h2 style="margin: 0; color: #38bdf8; font-size: 20px;">Wahy Transcriber is Live on Colab!</h2>
          <p style="margin: 0; color: #94a3b8; font-size: 13px;">Task 2: Abu-Dju Whisper Large V3 + Laya Envelope v3 (T4 GPU)</p>
        </div>
      </div>
      <div style="margin: 18px 0; background: #0284c7; padding: 14px 20px; border-radius: 12px; text-align: center;">
        <a href="{tunnel_url}" target="_blank" style="color: #ffffff; text-decoration: none; font-size: 16px; font-weight: bold;">
          👉 Click here to open Wahy Transcriber App 🔗
        </a>
      </div>
      <p style="font-size: 12px; color: #cbd5e1; margin: 0;">
        <strong>Public Link:</strong> <code style="background: #0f172a; padding: 2px 6px; border-radius: 4px; color: #38bdf8;">{tunnel_url}</code>
      </p>
    </div>
    """))
else:
    print("⚠️ Could not automatically detect Cloudflare URL. Check logs below:")

# 5. Keep cell running and print live logs
print("\n--- [Live Server Logs] ---")
try:
    while True:
        line = backend_proc.stdout.readline()
        if line:
            print(line, end="")
        else:
            time.sleep(0.5)
except KeyboardInterrupt:
    print("\n🛑 Stopping servers...")
    backend_proc.terminate()
    cf_proc.terminate()
    print("Server stopped.")
